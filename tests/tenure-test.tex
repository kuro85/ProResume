% -----------------------------------------------------------------------------
% Test Framework
% -----------------------------------------------------------------------------
\int_new:N \g__test_pass_int
\int_new:N \g__test_fail_int
\int_new:N \g__test_total_int
\str_new:N \l__test_actual_str
\str_new:N \l__test_expected_str

% Start a test section
\NewDocumentCommand{\testsection}{ m }{
  \iow_term:n { }
  \iow_term:n { ==~#1~== }
}

% Log present date header
\NewDocumentCommand{\presentheader}{ m }{
  \iow_term:n { }
  \iow_term:n { ============================================== }
  \iow_term:x { PRESENT~DATE:~#1 }
  \iow_term:n { ============================================== }
}

% Assert tenure calculation matches expected value
% #1 = start date, #2 = end date, #3 = expected tenure string
\NewDocumentCommand{\asserttenure}{ m m m }{
  \int_gincr:N \g__test_total_int
  \__tenure_calc:nn { #1 } { #2 }
  \str_set:Nx \l__test_actual_str { \tl_to_str:N \g__tenure_result_tl }
  \str_set:Nn \l__test_expected_str { #3 }
  \str_if_eq:NNTF \l__test_actual_str \l__test_expected_str {
    \int_gincr:N \g__test_pass_int
    \iow_term:x { PASS:~#1~to~#2~=>~"#3" }
  } {
    \int_gincr:N \g__test_fail_int
    \iow_term:x { FAIL:~#1~to~#2 }
    \iow_term:x { ~~expected:~"#3" }
    \iow_term:x { ~~actual:~~"\str_use:N \l__test_actual_str" }
  }
}

% Print test summary
\NewDocumentCommand{\testsummary}{}{
  \iow_term:n { }
  \iow_term:n { ========================================== }
  \iow_term:x { Test~Summary:~\int_use:N \g__test_pass_int /\int_use:N \g__test_total_int ~passed,~\int_use:N \g__test_fail_int ~failed }
  \iow_term:n { ========================================== }
  \int_compare:nNnTF { \g__test_fail_int } > { 0 } {
    \iow_term:n { TESTS~FAILED }
    \msg_fatal:nn { tenure-test } { tests-failed }
  } {
    \iow_term:n { ALL~TESTS~PASSED }
  }
}

\msg_new:nnn { tenure-test } { tests-failed } {
  \int_use:N \g__test_fail_int ~test(s)~failed.~See~output~above~for~details.
}
